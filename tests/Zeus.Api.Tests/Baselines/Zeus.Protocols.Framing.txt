    {
        public static byte[] Compute(Zeus.FrameChecksumKind kind, System.ReadOnlySpan<byte> data) { }
        public static ushort Crc16Modbus(System.ReadOnlySpan<byte> data) { }

    public enum FrameChecksumKind
    {
        None = 0,
        Xor8 = 1,
        Sum8 = 2,
        Crc16Modbus = 3,
    }
    public sealed class FrameLayout
    {
        public FrameLayout() { }
        public FrameLayout(byte[] header, Zeus.FrameLengthKind lengthKind, Zeus.FrameChecksumKind checksum) { }
        public Zeus.FrameChecksumKind Checksum { get; }
        public int ChecksumSize { get; }
        public System.Collections.Generic.IReadOnlyList<byte> Header { get; }
        public int LengthFieldSize { get; }
        public Zeus.FrameLengthKind LengthKind { get; }
        public int MaxPayloadLength { get; }
    }
    public enum FrameLengthKind
    {
        UInt8 = 0,
        UInt16LittleEndian = 1,
        UInt16BigEndian = 2,
    }

        public System.Threading.Tasks.Task SendAsync(System.ReadOnlyMemory<byte> payload, System.Threading.CancellationToken cancellationToken = default) { }
    }

    }
